cmake_minimum_required(VERSION 3.20)
project(devdocqa_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DOCQA_CORE_ONLY "Build dependency-free scheduler/configuration tests" OFF)
option(DOCQA_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(docqa_core INTERFACE)
target_include_directories(docqa_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
enable_testing()
add_executable(docqa_core_test tests/core_test.cpp)
target_link_libraries(docqa_core_test PRIVATE docqa_core)
add_test(NAME docqa_core COMMAND docqa_core_test)

if(NOT DOCQA_CORE_ONLY)
  find_package(Drogon CONFIG REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(devdocqa src/main.cpp src/server.cpp)
  target_link_libraries(devdocqa PRIVATE docqa_core Drogon::Drogon OpenSSL::Crypto Threads::Threads)
  install(TARGETS devdocqa RUNTIME DESTINATION bin)
endif()

foreach(target docqa_core_test devdocqa)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
      if(DOCQA_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
  endif()
endforeach()
